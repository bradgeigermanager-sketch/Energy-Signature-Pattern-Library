`js
const fs = require("fs");
const path = require("path");

const patternsDir = path.join(dirname, "patterns");
const manifestPath = path.join(dirname, "patterns_manifest.json");
const schemaPath = path.join(dirname, "schema/energy_signature.schema.json");

function emit(event, payload) {
  console.log(JSON.stringify({ event, payload, timestamp: new Date().toISOString() }));
}

function validatePattern(raw, schema) {
  const errors = [];

  if (!raw.id) errors.push("Missing id");
  if (!raw.name) errors.push("Missing name");
  if (!raw.geometry) errors.push("Missing geometry");
  if (!raw.fields) errors.push("Missing fields");

  return errors;
}

function run() {
  const schema = JSON.parse(fs.readFileSync(schemaPath, "utf8"));
  const files = fs.readdirSync(patternsDir).filter(f => f.endsWith(".json"));

  const manifest = {
    generatedAt: new Date().toISOString(),
    patterns: []
  };

  files.forEach(file => {
    const raw = JSON.parse(fs.readFileSync(path.join(patternsDir, file), "utf8"));
    const errors = validatePattern(raw, schema);

    if (errors.length) {
      emit("PatternValidationFailed", { file, errors });
      return;
    }

    manifest.patterns.push(file);
    emit("PatternAdded", { file });
  });

  fs.writeFileSync(manifestPath, JSON.stringify(manifest, null, 2));
  emit("ManifestUpdated", { manifestPath, count: manifest.patterns.length });
}

run();
`

This is now a DAMMS‑compliant agent that emits DAMMS events.

🧬 3. Agent Implementation (Node‑based)

Create:

`
patterneditoragent.js
`

`js
const fs = require("fs");
const path = require("path");

const patternsDir = path.join(dirname, "patterns");
const schemaPath = path.join(dirname, "schema/energy_signature.schema.json");

function emit(event, payload) {
  console.log(JSON.stringify({ event, payload, timestamp: new Date().toISOString() }));
}

function loadSchema() {
  return JSON.parse(fs.readFileSync(schemaPath, "utf8"));
}

function validatePattern(raw, schema) {
  const errors = [];

  if (!raw.id) errors.push("Missing id");
  if (!raw.name) errors.push("Missing name");
  if (!raw.geometry) errors.push("Missing geometry");
  if (!raw.fields) errors.push("Missing fields");

  return errors;
}

function createPattern({ id, name }) {
  const schema = loadSchema();

  const pattern = {
    id,
    name,
    description: "",
    geometry: schema.geometry.default || {},
    fields: schema.fields.default || {},
    particles: {},
    dynamo: {},
    thermodynamics: {},
    electrodynamics: {},
    metadata: { version: "1.0" }
  };

  const filePath = path.join(patternsDir, ${id}.json);
  fs.writeFileSync(filePath, JSON.stringify(pattern, null, 2));

  emit("PatternCreated", { id, filePath });
  emit("RequestManifestRegeneration", {});
}

function updatePattern({ id, patch }) {
  const filePath = path.join(patternsDir, ${id}.json);
  const raw = JSON.parse(fs.readFileSync(filePath, "utf8"));

  const updated = { ...raw, ...patch };

  const schema = loadSchema();
  const errors = validatePattern(updated, schema);
  if (errors.length) {
    emit("PatternValidationFailed", { id, errors });
    return;
  }

  const version = parseFloat(updated.metadata.version || "1.0");
  updated.metadata.version = (version + 0.1).toFixed(1);

  fs.writeFileSync(filePath, JSON.stringify(updated, null, 2));

  emit("PatternUpdated", { id, filePath });
  emit("RequestManifestRegeneration", {});
}

function deletePattern({ id }) {
  const filePath = path.join(patternsDir, ${id}.json);
  fs.unlinkSync(filePath);

  emit("PatternDeleted", { id });
  emit("RequestManifestRegeneration", {});
}

module.exports = {
  createPattern,
  updatePattern,
  deletePattern
};
`

This is a fully functional DAMMS agent.
