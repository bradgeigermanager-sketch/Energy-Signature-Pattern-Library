3. app.js

// ---------- Domain classes for MHD schema ----------

class ScalarHarmonic {
  constructor(raw) {
    this.modes = raw?.modes || [];
  }

  summary() {
    return ${this.modes.length} scalar modes;
  }
}

class FieldHarmonic {
  constructor(raw) {
    this.modes = raw?.modes || [];
  }

  summary() {
    return ${this.modes.length} vector modes;
  }
}

class MHDPattern {
  constructor(raw) {
    this.raw = raw;
    this.id = raw.id;
    this.name = raw.name;
    this.description = raw.description;

    this.geometry = raw.geometry || {};
    this.fields = {
      electricField: new FieldHarmonic(raw.fields?.electricField || {}),
      magneticField: new FieldHarmonic(raw.fields?.magneticField || {}),
      velocityField: new FieldHarmonic(raw.fields?.velocityField || {}),
      pressure: new ScalarHarmonic(raw.fields?.pressure || {}),
      temperature: new ScalarHarmonic(raw.fields?.temperature || {}),
      density: new ScalarHarmonic(raw.fields?.density || {})
    };

    this.particles = raw.particles || {};
    this.dynamo = raw.dynamo || {};
    this.thermodynamics = raw.thermodynamics || {};
    this.electrodynamics = {
      chargeDistribution: new ScalarHarmonic(raw.electrodynamics?.chargeDistribution || {}),
      currentDensity: new FieldHarmonic(raw.electrodynamics?.currentDensity || {}),
      radiationModel: raw.electrodynamics?.radiationModel || ""
    };

    this.metadata = raw.metadata || {};
  }
}

// ---------- State ----------

const state = {
  patterns: [],
  selectedPattern: null,
  selectedFieldKey: "magneticField"
};

// ---------- DOM helpers ----------

function $(sel) {
  return document.querySelector(sel);
}

function createEl(tag, className, text) {
  const el = document.createElement(tag);
  if (className) el.className = className;
  if (text !== undefined) el.textContent = text;
  return el;
}

// ---------- Rendering: pattern list ----------

function renderPatternList() {
  const container = $("#pattern-list");
  container.innerHTML = "";

  if (!state.patterns.length) {
    container.textContent = "No patterns loaded.";
    return;
  }

  state.patterns.forEach(p => {
    const item = createEl("div", "pattern-item");
    item.dataset.id = p.id;

    if (state.selectedPattern && state.selectedPattern.id === p.id) {
      item.classList.add("active");
    }

    const nameEl = createEl("div", "pattern-name", p.name);
    const metaEl = createEl(
      "div",
      "pattern-meta",
      radius=${p.geometry.radius ?? "?"}, lMax=${p.geometry.harmonicBasis?.lMax ?? "?"}
    );

    item.appendChild(nameEl);
    item.appendChild(metaEl);

    item.addEventListener("click", () => {
      state.selectedPattern = p;
      renderPatternList();
      renderPatternOverview();
      renderGeometry();
      renderParticles();
      renderDynamo();
      renderThermo();
      renderElectro();
      renderFieldTabs();
      renderFieldModes();
    });

    container.appendChild(item);
  });
}

// ---------- Rendering: overview & blocks ----------

function renderPatternOverview() {
  const container = $("#pattern-overview");
  container.innerHTML = "";

  if (!state.selectedPattern) {
    container.innerHTML = "<p class='small-muted'>Select a pattern to view details.</p>";
    return;
  }

  const p = state.selectedPattern;

  const descBlock = createEl("div", "detail-block");
  const label = createEl("div", "detail-label", "Description");
  const text = createEl("div", "detail-text", p.description || "No description.");
  descBlock.appendChild(label);
  descBlock.appendChild(text);

  const metaBlock = createEl("div", "detail-block");
  const metaLabel = createEl("div", "detail-label", "Metadata");
  const metaCode = createEl("pre", "code-block");
  metaCode.textContent = JSON.stringify(p.metadata, null, 2);
  metaBlock.appendChild(metaLabel);
  metaBlock.appendChild(metaCode);

  container.appendChild(descBlock);
  container.appendChild(metaBlock);
}

function renderGeometry() {
  const container = $("#geometry-detail");

  if (!state.selectedPattern) {

  const code = createEl("pre", "code-block");
  code.textContent = JSON.stringify(g, null, 2);

  block.appendChild(label);
  block.appendChild(code);
  container.appendChild(block);
}

// ---------- Rendering: fields & modes ----------

const FIELD_KEYS = [
  { key: "magneticField", label: "Magnetic field B" },
  { key: "electricField", label: "Electric field E" },
  { key: "velocityField", label: "Velocity field v" },
  { key: "pressure", label: "Pressure p" },
  { key: "temperature", label: "Temperature T" },
  { key: "density", label: "Density ρ" },
  { key: "chargeDistribution", label: "Charge distribution ρ_q" },
  { key: "currentDensity", label: "Current density J" }
];

function getFieldObject(pattern, key) {
  if (key === "chargeDistribution") return pattern.electrodynamics.chargeDistribution;
  if (key === "currentDensity") return pattern.electrodynamics.currentDensity;
  return pattern.fields[key];
}

function renderFieldTabs() {
  const container = $("#field-tabs");
  container.innerHTML = "";

  if (!state.selectedPattern) {
    container.innerHTML = "<p class='small-muted'>No pattern selected.</p>";
    return;
  }

  FIELD_KEYS.forEach(f => {
    const tab = createEl("button", "field-tab", f.label);
    if (state.selectedFieldKey === f.key) {
      tab.classList.add("active");
    }
    tab.addEventListener("click", () => {
      state.selectedFieldKey = f.key;
      renderFieldTabs();
      renderFieldModes();
    });
    container.appendChild(tab);
  });
}

function renderFieldModes() {
  const container = $("#field-modes");
  container.innerHTML = "";

  if (!state.selectedPattern) {
    container.innerHTML = "<p class='small-muted'>No pattern selected.</p>";
    return;
  }

  const fieldObj = getFieldObject(state.selectedPattern, state.selectedFieldKey);
  if (!fieldObj || !fieldObj.modes || !fieldObj.modes.length) {
    container.innerHTML = "<p class='small-muted'>No modes defined for this field.</p>";
    return;
  }

  const list = createEl("div", "mode-list");

  fieldObj.modes.forEach((mode, idx) => {
    const item = createEl("div", "mode-item");

    const header = createEl("div", "mode-header");
    const label = createEl(
      "div",
      "mode-label",
      Mode ${idx + 1}: l=${mode.l}, m=${mode.m}
    );
    const tag = createEl(
      "div",
      "mode-tag",
      mode.vectorComponents ? "vector harmonic" : "scalar harmonic"
    );
    header.appendChild(label);
    header.appendChild(tag);

    const body = createEl("div", "mode-body");

    const rp = createEl("div", null);
    rp.innerHTML = <strong>Radial profile:</strong> ${mode.radialProfile || "—"};
    body.appendChild(rp);

    if (mode.vectorComponents) {
      const vc = mode.vectorComponents;
      const vcDiv = createEl("div", null);
      vcDiv.innerHTML =
        <strong>Components:</strong>  +
        <span class="inline-pill">E<sub>r</sub> / B<sub>r</sub> / v<sub>r</sub></span> ${vc.r || "0"},  +
        <span class="inline-pill">θ</span> ${vc.theta || "0"},  +
        <span class="inline-pill">φ</span> ${vc.phi || "0"};
      body.appendChild(vcDiv);
    } else if (typeof mode.coefficient !== "undefined") {
      const coeffDiv = createEl("div", null);
      coeffDiv.innerHTML = <strong>Coefficient:</strong> ${mode.coefficient};
      body.appendChild(coeffDiv);
    }

    item.appendChild(header);
    item.appendChild(body);
    list.appendChild(item);
  });

  container.appendChild(list);
}

async function loadPatternLibrary() {
  try {
    const res = await fetch("patterns_mhd.json");
    const arr = await res.json();

    state.patterns = arr.map(raw => new MHDPattern(raw));
    state.selectedPattern = state.patterns[0];

    renderPatternList();
    renderPatternOverview();
    renderGeometry();
    renderParticles();
    renderDynamo();

  } catch (err) {
    console.error("Failed to load patterns_mhd.json", err);

  }

Now the interface loads all patterns from the JSON file.

🧪 3. Add schema‑aware validation (optional but recommended)

Insert this helper:

`js
function validatePattern(raw) {
  const errors = [];

  if (!raw.id) errors.push("Missing id");
  if (!raw.name) errors.push("Missing name");
  if (!raw.geometry) errors.push("Missing geometry");
  if (!raw.fields) errors.push("Missing fields");

  if (raw.geometry && raw.geometry.harmonicBasis) {
    const hb = raw.geometry.harmonicBasis;
    if (hb.lMax < 0) errors.push("lMax must be >= 0");
    if (hb.mMax < 0) errors.push("mMax must be >= 0");
  }

  return errors;
}
`

Then modify the loader:

`js
state.patterns = arr.map(raw => {
  const errs = validatePattern(raw);
  if (errs.length) {
    console.warn(Pattern ${raw.id} has validation errors:, errs);
  }
  return new MHDPattern(raw);
});
`

This gives you schema‑level safety without needing a full JSON Schema validator.

⚙️ 3. Updated Loader in app.js

Replace your loader with this multi‑file loader:

`js
async function loadPatternLibrary() {
  try {
    // Load manifest
    const manifestRes = await fetch("patterns_manifest.json");
    const manifest = await manifestRes.json();

    const patternFiles = manifest.patterns || [];
    const loaded = [];

    // Load each pattern file individually
    for (const file of patternFiles) {
      try {
        const res = await fetch(patterns/${file});
        const raw = await res.json();
        loaded.push(new MHDPattern(raw));
      } catch (err) {
        console.warn(Failed to load pattern file: ${file}, err);
      }
    }

    state.patterns = loaded;
    state.selectedPattern = loaded[0] || null;

    renderPatternList();
    renderPatternOverview();
    renderGeometry();
    renderParticles();
    renderDynamo();
    renderThermo();
    renderElectro();
    renderFieldTabs();
    renderFieldModes();

  } catch (err) {
    console.error("Failed to load manifest", err);
    $("#pattern-list").textContent = "Error loading pattern manifest.";
  }
}
`

This loader:

- Reads the manifest  
- Loads each pattern file individually  
- Wraps each in your MHDPattern class  
- Populates the UI
